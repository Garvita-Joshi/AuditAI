"""Leakage-safe AuditAI anomaly-detection pipeline.

This version deliberately rebuilds time-dependent features *after sorting by
transaction timestamp*.  For every row, aggregate features use only records
that occurred earlier.  It does not trust precomputed aggregate columns in an
input CSV, because those are commonly calculated over the full data set.

Expected raw fields: transaction_date, amount, vendor_id, employee_id and,
when used, department_id/department_name, transaction_time,
vendor_registration_date and approval_limit.  Set DATA_PATH before running.
"""
from __future__ import annotations

import copy
import random
from pathlib import Path

import joblib
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
import seaborn as sns
import torch
import torch.nn as nn
from sklearn.compose import ColumnTransformer
from sklearn.ensemble import IsolationForest
from sklearn.impute import SimpleImputer
from sklearn.metrics import (ConfusionMatrixDisplay, PrecisionRecallDisplay,
                             RocCurveDisplay, average_precision_score,
                             classification_report, confusion_matrix, f1_score,
                             precision_score, recall_score, roc_auc_score)
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler


# ------------------------------- Configuration ----------------------------
DATA_PATH = None  # Set a full path, or leave None for Colab upload/search.
DATASET_FILENAME = "audit_anomalies_joined_dataset.csv"
TARGET = "is_anomaly"  # Synthetic benchmark label; never an ML input.
# Labels were deliberately generated with the synthetic anomalies, so they may
# calibrate the validation threshold and evaluate the untouched test period.
LABELS_ARE_INDEPENDENTLY_VALIDATED = True
MAX_ONE_HOT_CARDINALITY = 30
MASTER_DATA_AS_OF_SAFE = False

SEED = 42
EPOCHS, BATCH_SIZE, LEARNING_RATE, LATENT_DIM = 100, 256, 1e-3, 16
PATIENCE, NORMAL_THRESHOLD_QUANTILE = 12, 0.995
AE_WEIGHT, IF_WEIGHT, ISOLATION_CONTAMINATION = 0.50, 0.50, 0.05
DEVICE = "cuda" if torch.cuda.is_available() else "cpu"

random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)


def locate_dataset() -> Path:
    """Find the CSV in Colab/Drive, or open Colab's normal upload control."""
    if DATA_PATH:
        explicit = Path(DATA_PATH)
        if explicit.is_file():
            return explicit
        print(f"Configured DATA_PATH was not found: {explicit}")
    for candidate in (Path("/content") / DATASET_FILENAME,
                      Path("/content/drive/MyDrive") / DATASET_FILENAME):
        if candidate.is_file():
            return candidate
    drive_root = Path("/content/drive/MyDrive")
    if drive_root.is_dir():
        matches = list(drive_root.rglob(DATASET_FILENAME))
        if matches:
            return matches[0]
    try:
        from google.colab import files
        print(f"Select {DATASET_FILENAME} in the upload dialog.")
        uploaded = files.upload()
        for name in uploaded:
            path = Path("/content") / name
            if path.suffix.lower() == ".csv":
                return path
    except ImportError as exc:
        raise FileNotFoundError(
            f"Could not find {DATASET_FILENAME}. Set DATA_PATH to its full path."
        ) from exc
    raise FileNotFoundError("No CSV was uploaded.")


def first_existing(frame: pd.DataFrame, *names: str) -> str | None:
    """Return the first available column name from a list of aliases."""
    return next((name for name in names if name in frame.columns), None)


def prior_group_count(frame: pd.DataFrame, group: str) -> pd.Series:
    """Number of earlier rows in a group; the current row is deliberately excluded."""
    return frame.groupby(group, dropna=False).cumcount().astype("float32")


def prior_group_mean(frame: pd.DataFrame, group: str, value: str) -> pd.Series:
    """Mean value of earlier rows in a group, never including the current row."""
    previous_sum = frame.groupby(group, dropna=False)[value].cumsum() - frame[value]
    previous_count = prior_group_count(frame, group)
    return previous_sum / previous_count.replace(0, np.nan)


def build_point_in_time_features(raw: pd.DataFrame) -> pd.DataFrame:
    """Return one feature row per transaction with no future transaction data.

    Sorting uses timestamp plus the original row order.  For simultaneous events,
    their input order becomes the event order; provide a real event sequence/id
    upstream if transactions can share a timestamp.
    """
    df = raw.copy()
    required = {"transaction_date", "amount", "vendor_id", "employee_id"}
    absent = required - set(df.columns)
    if absent:
        raise ValueError(f"Missing required raw columns: {sorted(absent)}")

    date = pd.to_datetime(df["transaction_date"], errors="coerce")
    time_col = first_existing(df, "transaction_time", "time")
    if time_col:
        time = pd.to_timedelta(df[time_col].astype(str), errors="coerce").fillna(pd.Timedelta(0))
    else:
        time = pd.Timedelta(0)
    df["_event_time"] = date + time
    df["_source_order"] = np.arange(len(df))
    df["amount"] = pd.to_numeric(df["amount"], errors="coerce")
    df = df.dropna(subset=["_event_time", "amount"]).sort_values(["_event_time", "_source_order"], kind="stable").reset_index(drop=True)

    # Calendar/current-transaction features: available as soon as it is entered.
    df["hour"] = df["_event_time"].dt.hour.astype("float32")
    df["day_of_week"] = df["_event_time"].dt.dayofweek.astype("float32")
    df["month"] = df["_event_time"].dt.month.astype("float32")
    df["quarter"] = df["_event_time"].dt.quarter.astype("float32")
    df["weekend_flag"] = (df["day_of_week"] >= 5).astype("float32")
    df["log_amount"] = np.log1p(df["amount"].clip(lower=0))

    # Historical global baseline.  shift-by-construction prevents self leakage.
    old_count = pd.Series(np.arange(len(df)), index=df.index, dtype="float64")
    old_sum = df["amount"].cumsum() - df["amount"]
    old_mean = old_sum / old_count.replace(0, np.nan)
    old_sq_sum = (df["amount"] ** 2).cumsum() - df["amount"] ** 2
    old_var = (old_sq_sum / old_count.replace(0, np.nan) - old_mean ** 2).clip(lower=0)
    df["amount_zscore"] = ((df["amount"] - old_mean) / np.sqrt(old_var.replace(0, np.nan))).replace([np.inf, -np.inf], np.nan)

    df["vendor_transaction_count"] = prior_group_count(df, "vendor_id")
    df["vendor_avg_amount"] = prior_group_mean(df, "vendor_id", "amount")
    df["amount_to_vendor_avg_ratio"] = df["amount"] / df["vendor_avg_amount"].replace(0, np.nan)

    department = first_existing(df, "department_id", "department_name")
    if department:
        department_avg = prior_group_mean(df, department, "amount")
        df["amount_to_department_avg_ratio"] = df["amount"] / department_avg.replace(0, np.nan)

    df["employee_transaction_count"] = prior_group_count(df, "employee_id")
    df["employee_avg_amount"] = prior_group_mean(df, "employee_id", "amount")

    # Month-to-date *prior* frequency; not the completed-month total.
    df["_year_month"] = df["_event_time"].dt.to_period("M").astype(str)
    df["vendor_monthly_frequency"] = df.groupby(["vendor_id", "_year_month"], dropna=False).cumcount().astype("float32")

    registration = first_existing(df, "vendor_registration_date", "vendor_registered_at")
    if registration:
        registered_at = pd.to_datetime(df[registration], errors="coerce")
        df["vendor_days_since_registration"] = (df["_event_time"].dt.normalize() - registered_at.dt.normalize()).dt.days.clip(lower=0)

    # Include only values known under an effective-dated approval policy.
    approval_limit = first_existing(df, "approval_limit", "approver_limit")
    if approval_limit:
        limit = pd.to_numeric(df[approval_limit], errors="coerce")
        df["approval_limit_ratio"] = df["amount"] / limit.replace(0, np.nan)

    # These counts must come from a master-data snapshot valid on _event_time.
    # Existing CSV values are intentionally ignored unless that provenance is known.
    if MASTER_DATA_AS_OF_SAFE:
        for col in ("shared_phone_count", "shared_address_count", "shared_bank_account_count"):
            if col in df:
                df[col] = pd.to_numeric(df[col], errors="coerce")

    return df.drop(columns=["_source_order"], errors="ignore")


def chronological_split(frame: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    n = len(frame)
    if n < 50:
        raise ValueError("At least 50 timestamped transactions are required.")
    first, second = int(n * 0.70), int(n * 0.85)
    return frame.iloc[:first].copy(), frame.iloc[first:second].copy(), frame.iloc[second:].copy()


class TransactionAutoencoder(nn.Module):
    def __init__(self, input_dim: int):
        super().__init__()
        h1, h2 = max(64, min(256, input_dim * 2)), max(32, min(128, input_dim))
        self.encoder = nn.Sequential(nn.Linear(input_dim, h1), nn.ReLU(), nn.Dropout(0.10), nn.Linear(h1, h2), nn.ReLU(), nn.Linear(h2, LATENT_DIM))
        self.decoder = nn.Sequential(nn.Linear(LATENT_DIM, h2), nn.ReLU(), nn.Linear(h2, h1), nn.ReLU(), nn.Linear(h1, input_dim))

    def forward(self, x):
        return self.decoder(self.encoder(x))


def reconstruction_errors(model: nn.Module, x: np.ndarray) -> np.ndarray:
    model.eval(); result = []
    with torch.no_grad():
        for start in range(0, len(x), BATCH_SIZE):
            batch = torch.as_tensor(x[start:start + BATCH_SIZE], dtype=torch.float32, device=DEVICE)
            result.append(((model(batch) - batch) ** 2).mean(dim=1).cpu().numpy())
    return np.concatenate(result)


def normalize_from_normal(values: np.ndarray, normal_reference: np.ndarray) -> np.ndarray:
    low, high = np.quantile(normal_reference, [0.01, 0.995])
    return np.clip((values - low) / max(high - low, 1e-9), 0, 1)


def main() -> None:
    dataset_path = locate_dataset()
    print("Using device:", DEVICE)
    print("Dataset path:", dataset_path)
    raw = pd.read_csv(dataset_path, low_memory=False)
    frame = build_point_in_time_features(raw)
    train, validation, test = chronological_split(frame)
    if TARGET not in frame:
        raise ValueError(f"{TARGET!r} is required to identify normal training rows.")
    normal_train = train[train[TARGET].eq(0)].copy()
    normal_validation = validation[validation[TARGET].eq(0)].copy()
    if normal_train.empty or normal_validation.empty:
        raise ValueError("Training and validation periods each need known-normal transactions.")

    # Do not model transaction_id, raw date/time, or human/rule-generated outputs.
    numeric = [
        "amount", "log_amount", "amount_zscore", "amount_to_vendor_avg_ratio", "amount_to_department_avg_ratio",
        "hour", "day_of_week", "month", "quarter", "weekend_flag", "vendor_transaction_count",
        "vendor_avg_amount", "vendor_days_since_registration", "vendor_monthly_frequency",
        "employee_transaction_count", "employee_avg_amount", "approval_limit_ratio",
        "shared_phone_count", "shared_address_count", "shared_bank_account_count",
    ]
    # IDs are deliberately excluded: one-hot encoding vendor/employee/approver
    # identifiers lets the AE memorize identities. Historical behavioural
    # features above retain their useful information without high dimension.
    categorical_candidates = [
        "entry_type", "general_ledger_account", "currency", "payment_mode",
        "department_name",
    ]
    numeric = [c for c in numeric if c in frame]
    available_categorical = [c for c in categorical_candidates if c in frame]
    # Cardinality is measured on normal *training* rows only: validation/test
    # values never influence which categories are encoded.
    cardinality = {c: int(normal_train[c].nunique(dropna=True)) for c in available_categorical}
    categorical = [c for c in available_categorical if cardinality[c] <= MAX_ONE_HOT_CARDINALITY]
    excluded_categorical = [c for c in available_categorical if c not in categorical]
    if not numeric and not categorical:
        raise ValueError("No usable model features were built.")
    print("Using numeric features:", numeric)
    print("One-hot categorical features (training cardinality <= "
          f"{MAX_ONE_HOT_CARDINALITY}):", categorical)
    if excluded_categorical:
        print("Excluded high-cardinality categorical features:", excluded_categorical)

    preprocessor = ColumnTransformer([
        ("numeric", Pipeline([("impute", SimpleImputer(strategy="median", add_indicator=True)), ("scale", StandardScaler())]), numeric),
        ("categorical", Pipeline([("impute", SimpleImputer(strategy="most_frequent")), ("onehot", OneHotEncoder(handle_unknown="ignore", min_frequency=5, sparse_output=False))]), categorical),
    ])
    x_train = preprocessor.fit_transform(normal_train).astype("float32")
    x_val_normal = preprocessor.transform(normal_validation).astype("float32")
    x_val = preprocessor.transform(validation).astype("float32")
    x_test = preprocessor.transform(test).astype("float32")
    print("\n===== FEATURE DIMENSIONALITY =====")
    print(f"Original selected input columns: {len(numeric) + len(categorical)}")
    print(f"Numerical features: {len(numeric)}")
    print(f"Categorical features one-hot encoded: {len(categorical)}")
    print(f"Final features after preprocessing: {x_train.shape[1]}")

    model = TransactionAutoencoder(x_train.shape[1]).to(DEVICE)
    optimizer, loss_fn = torch.optim.AdamW(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-5), nn.MSELoss()
    train_tensor = torch.as_tensor(x_train, dtype=torch.float32, device=DEVICE)
    best_state, best_val_loss, waiting = None, float("inf"), 0
    history = {"train_loss": [], "validation_normal_loss": []}
    for epoch in range(1, EPOCHS + 1):
        model.train(); losses = []
        for indices in torch.randperm(len(train_tensor), device=DEVICE).split(BATCH_SIZE):
            batch = train_tensor[indices]; optimizer.zero_grad(); loss = loss_fn(model(batch), batch)
            loss.backward(); optimizer.step(); losses.append(loss.item())
        val_loss = float(reconstruction_errors(model, x_val_normal).mean())
        history["train_loss"].append(float(np.mean(losses)))
        history["validation_normal_loss"].append(val_loss)
        if val_loss < best_val_loss:
            best_val_loss, best_state, waiting = val_loss, copy.deepcopy(model.state_dict()), 0
        else:
            waiting += 1
        if epoch == 1 or epoch % 10 == 0: print(f"epoch={epoch:03d} train_loss={np.mean(losses):.6f} val_normal_loss={val_loss:.6f}")
        if waiting >= PATIENCE: break
    model.load_state_dict(best_state)

    # Graph 1: validation loss governs early stopping; training loss alone does not.
    plt.figure(figsize=(9, 4.5))
    plt.plot(history["train_loss"], label="Normal training loss")
    plt.plot(history["validation_normal_loss"], label="Unseen normal validation loss")
    plt.xlabel("Epoch"); plt.ylabel("Reconstruction MSE")
    plt.title("Autoencoder training history")
    plt.legend(); plt.grid(alpha=0.25); plt.tight_layout(); plt.show()

    # Fit IF on original normal vectors: not on AE reconstructions.
    forest = IsolationForest(n_estimators=300, contamination=ISOLATION_CONTAMINATION, random_state=SEED, n_jobs=-1).fit(x_train)
    ae_train, if_train = reconstruction_errors(model, x_train), -forest.decision_function(x_train)
    def score(x):
        ae, iso = reconstruction_errors(model, x), -forest.decision_function(x)
        combined = AE_WEIGHT * normalize_from_normal(ae, ae_train) + IF_WEIGHT * normalize_from_normal(iso, if_train)
        return combined, ae, iso
    val_score, _, _ = score(x_val)
    test_score, test_ae, test_if = score(x_test)

    # Synthetic labels calibrate only the validation decision boundary. They are
    # never a model input and the test labels are not read until final evaluation.
    validation_y = validation[TARGET].to_numpy()
    candidate_thresholds = np.unique(val_score)
    validation_f1 = np.asarray([f1_score(validation_y, val_score >= t, zero_division=0)
                                for t in candidate_thresholds])
    BEST_THRESHOLD = float(candidate_thresholds[np.argmax(validation_f1)])
    validation_prediction = (val_score >= BEST_THRESHOLD).astype(int)
    validation_precision = precision_score(validation_y, validation_prediction, zero_division=0)
    validation_recall = recall_score(validation_y, validation_prediction, zero_division=0)
    validation_f1_score = f1_score(validation_y, validation_prediction, zero_division=0)
    print("\n===== VALIDATION THRESHOLD CALIBRATION =====")
    print(f"Best Threshold: {BEST_THRESHOLD:.6f}")
    print(f"Validation Precision: {validation_precision:.3f}")
    print(f"Validation Recall:    {validation_recall:.3f}")
    print(f"Validation F1:        {validation_f1_score:.3f}")

    # This is frozen before test evaluation. Test labels never select it.
    threshold = BEST_THRESHOLD
    prediction = (test_score >= BEST_THRESHOLD).astype(int)

    # Graph 2: score distribution and the threshold used for audit triage.
    plt.figure(figsize=(9, 4.5))
    sns.histplot(test_score, bins=40, stat="density", color="#4c78a8", label="All test transactions")
    if TARGET in test:
        sns.kdeplot(data=test, x=test_score, hue=TARGET, common_norm=False, palette={0: "#54a24b", 1: "#e45756"}, warn_singular=False)
    plt.axvline(threshold, color="black", linestyle="--", label=f"Threshold = {threshold:.3f}")
    plt.xlabel("Combined anomaly score"); plt.title("Test-period anomaly-score distribution")
    plt.legend(); plt.tight_layout(); plt.show()

    if LABELS_ARE_INDEPENDENTLY_VALIDATED:
        y_test = test[TARGET].to_numpy()
        print("\n===== FINAL TEST EVALUATION =====")
        print(f"Test Precision: {precision_score(y_test, prediction, zero_division=0):.3f}")
        print(f"Test Recall:    {recall_score(y_test, prediction, zero_division=0):.3f}")
        print(f"Test F1:        {f1_score(y_test, prediction, zero_division=0):.3f}")
        print(f"Test PR-AUC:    {average_precision_score(y_test, test_score):.3f}")
        print(f"Test ROC-AUC:   {roc_auc_score(y_test, test_score):.3f}")
        print("\n", classification_report(y_test, prediction, target_names=["Normal", "Synthetic anomaly"], zero_division=0))
        print("\nInterpretation: these metrics are measured against intentionally generated "
              "synthetic anomaly labels. They are a controlled synthetic benchmark, "
              "not real-world fraud-detection accuracy.")

        # Graph 3: final evaluation on the untouched test period.
        pr_auc, roc_auc = average_precision_score(y_test, test_score), roc_auc_score(y_test, test_score)
        fig, axes = plt.subplots(1, 3, figsize=(18, 5))
        ConfusionMatrixDisplay(confusion_matrix(y_test, prediction), display_labels=["Normal", "Anomaly"]).plot(ax=axes[0], colorbar=False)
        axes[0].set_title("Confusion matrix")
        PrecisionRecallDisplay.from_predictions(y_test, test_score, ax=axes[1])
        axes[1].set_title(f"Precision–recall curve (AP={pr_auc:.3f})")
        RocCurveDisplay.from_predictions(y_test, test_score, ax=axes[2])
        axes[2].set_title(f"ROC curve (AUC={roc_auc:.3f})")
        plt.tight_layout(); plt.show()
    else:
        print("Metrics/label-based graphs suppressed: is_anomaly is not declared independently validated ground truth.")

    output = Path("auditai_model_artifacts"); output.mkdir(exist_ok=True)
    torch.save({"state_dict": model.state_dict(), "input_dim": x_train.shape[1], "latent_dim": LATENT_DIM, "threshold": threshold}, output / "autoencoder.pt")
    joblib.dump(preprocessor, output / "preprocessor.joblib"); joblib.dump(forest, output / "isolation_forest.joblib")
    review = test[[c for c in ["transaction_id", "transaction_date", "vendor_id", "employee_id", "amount", TARGET] if c in test]].copy()
    review["autoencoder_error"], review["isolation_forest_score"], review["combined_anomaly_score"], review["model_prediction"] = test_ae, test_if, test_score, prediction
    review = review.sort_values("combined_anomaly_score", ascending=False)
    review.to_csv(output / "test_set_model_review.csv", index=False)
    print("\nTop 30 transactions for audit review:")
    print(review.head(30).to_string(index=False))
    print(f"\nSaved model artifacts and review CSV to: {output.resolve()}")


if __name__ == "__main__":
    main()

## Synthetic-benchmark interpretation

`is_anomaly` is used only to calibrate the decision threshold on the validation period and to evaluate the untouched test period. It is never included in model features or used to train the Autoencoder or Isolation Forest. Results are performance on a controlled synthetic benchmark, not evidence of real-world fraud-detection accuracy.
