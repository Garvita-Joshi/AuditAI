"""AuditAI final pipeline: ablation of 4 feature recipes + all validation checks (Colab-ready).

WHAT IT DOES (one run, one identical chronological split for every variant)
  1. Audits the pre-computed aggregate columns in your CSV for full-ledger leakage.
  2. Rebuilds history features point-in-time and PROVES it with a "corrupt the future" test.
  3. Trains Autoencoder + Isolation Forest (normal rows only) for 4 feature variants x N seeds:
        A  = Untitled9-style point-in-time features
        B  = A + OCR / receipt features (Untitled14 had these, Untitled9 dropped them)
        C  = B + robust features (capped counts, log-scaled z-scores, new-entity flags, clipping)
        D  = C + raw vendor/employee/approver IDs   -> memorization check
        E  = supervised gradient boosting on C       -> reference ceiling (uses labels)
  4. Reports ROC-AUC / PR-AUC / precision / recall / F1 (mean +- std over seeds), loss curves
     (train vs unseen-normal), score distributions, confusion matrix, PR/ROC, alert-budget curve,
     per-risk_category recall, seed stability, drift (PSI), and a review CSV with top drivers.

NOTE ON LABELS: is_anomaly is a SYNTHETIC benchmark label. Every metric below measures agreement
with that generator, not real-world fraud-detection accuracy. Risk/label columns are never inputs.
"""

def set_seed(seed: int) -> None:
    random.seed(seed); np.random.seed(seed); torch.manual_seed(seed)

class TransactionAutoencoder(nn.Module):
    def __init__(self, input_dim: int):
        super().__init__()
        h1, h2 = max(64, min(256, input_dim * 2)), max(32, min(128, input_dim))
        self.encoder = nn.Sequential(nn.Linear(input_dim, h1), nn.ReLU(), nn.Dropout(0.10),
                                     nn.Linear(h1, h2), nn.ReLU(), nn.Linear(h2, LATENT_DIM))
        self.decoder = nn.Sequential(nn.Linear(LATENT_DIM, h2), nn.ReLU(), nn.Linear(h2, h1), nn.ReLU(),
                                     nn.Linear(h1, input_dim))
    def forward(self, x):
        return self.decoder(self.encoder(x))

class TrainedAE:
    """Wrapper with a framework-neutral interface: errors(), feature_errors(), save()."""
    def __init__(self, model, history):
        self.model, self.history = model, history
    def _sq_err(self, x: np.ndarray) -> np.ndarray:
        self.model.eval(); out = []
        with torch.no_grad():
            for s in range(0, len(x), BATCH_SIZE):
                b = torch.as_tensor(x[s:s + BATCH_SIZE], dtype=torch.float32, device=DEVICE)
                out.append(((self.model(b) - b) ** 2).cpu().numpy())
        return np.vstack(out)
    def errors(self, x): return self._sq_err(x).mean(axis=1)
    def feature_errors(self, x): return self._sq_err(x)
    def save(self, path, extra): torch.save({"state_dict": self.model.state_dict(), **extra}, path)

def train_autoencoder(x_train: np.ndarray, x_val_normal: np.ndarray, seed: int) -> TrainedAE:
    set_seed(seed)
    model = TransactionAutoencoder(x_train.shape[1]).to(DEVICE)
    opt, loss_fn = torch.optim.AdamW(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-5), nn.MSELoss()
    tr = torch.as_tensor(x_train, dtype=torch.float32, device=DEVICE)
    holder = TrainedAE(model, {"train_loss": [], "val_normal_loss": [], "best_epoch": 0})
    best, best_state, wait = float("inf"), None, 0
    for epoch in range(1, EPOCHS + 1):
        model.train(); losses = []
        for idx in torch.randperm(len(tr), device=DEVICE).split(BATCH_SIZE):
            batch = tr[idx]; opt.zero_grad(); loss = loss_fn(model(batch), batch); loss.backward(); opt.step()
            losses.append(loss.item())
        val = float(holder.errors(x_val_normal).mean())
        holder.history["train_loss"].append(float(np.mean(losses))); holder.history["val_normal_loss"].append(val)
        if val < best: best, best_state, wait, holder.history["best_epoch"] = val, copy.deepcopy(model.state_dict()), 0, epoch
        else: wait += 1
        if wait >= PATIENCE: break
    model.load_state_dict(best_state)
    return holder

def static_share(df, col, key):
    """Share of groups (>=2 rows) where `col` has ONE value. ~100% => whole-ledger snapshot, not a running value."""
    sub = df[df.groupby(key)[col].transform("size") >= 2]
    return float((sub.groupby(key)[col].nunique() <= 1).mean())

# %% 3. Point-in-time (leakage-safe) feature builder + "corrupt the future" proof
def first_existing(df, *names):
    return next((n for n in names if n in df.columns), None)

def prior_stats(df, keys, col):
    """count / mean / std of `col` over EARLIER rows of the same group (current row excluded)."""
    x = df[col]; g = df.groupby(keys, sort=False)
    n = g.cumcount(); s1 = g[col].cumsum() - x
    sq = df.assign(_sq=x ** 2); s2 = sq.groupby(keys, sort=False)["_sq"].cumsum() - sq["_sq"]
    nn = n.replace(0, np.nan); mean = s1 / nn
    return n, mean, np.sqrt((s2 / nn - mean ** 2).clip(lower=0))

def prior_window_count(df, key, days=30):
    """Earlier rows (by position) of the same group inside the previous `days` days."""
    out = np.zeros(len(df)); t = df["_event_time"].values.astype("datetime64[ns]")
    for idx in df.groupby(key, sort=False).indices.values():
        tg = t[idx]; out[idx] = np.arange(len(idx)) - np.searchsorted(tg, tg - np.timedelta64(days, "D"), side="right")
    return out

def build_features(raw_df: pd.DataFrame) -> pd.DataFrame:
    df = raw_df.copy()
    date = pd.to_datetime(df["transaction_date"], errors="coerce")
    tcol = first_existing(df, "transaction_time", "time")
    time = pd.to_timedelta(df[tcol].astype(str), errors="coerce").fillna(pd.Timedelta(0)) if tcol else pd.Timedelta(0)
    df["_event_time"], df["_source_order"] = date + time, np.arange(len(df))
    df["amount"] = pd.to_numeric(df["amount"], errors="coerce")
    df = (df.dropna(subset=["_event_time", "amount"])
            .sort_values(["_event_time", "_source_order"], kind="stable").reset_index(drop=True))
    # keys without NaN so "missing vendor" is its own group instead of silently vanishing
    df["_v"] = df["vendor_id"].fillna("__MISSING__").astype(str)
    df["_e"] = df["employee_id"].fillna("__MISSING__").astype(str)
    dcol = first_existing(df, "department_id", "department_name")
    df["_d"] = df[dcol].fillna("__MISSING__").astype(str) if dcol else "all"
    df["_all"] = 0
    ts = df["_event_time"]; df["_la"] = np.log1p(df["amount"].abs())

    # ---- calendar / row-level (known the moment the row is entered)
    df["hour"] = ts.dt.hour.astype("float32"); df["day_of_week"] = ts.dt.dayofweek.astype("float32")
    df["month"] = ts.dt.month.astype("float32"); df["quarter"] = ts.dt.quarter.astype("float32")
    df["weekend_flag"] = (df["day_of_week"] >= 5).astype("float32")
    df["is_after_hours"] = ((df["hour"] < 8) | (df["hour"] >= 20)).astype("float32")
    df["log_amount"] = np.log1p(df["amount"].clip(lower=0))
    df["is_negative"] = (df["amount"] < 0).astype("float32")
    df["is_round"] = ((df["amount"].abs() >= 10_000) & (df["amount"].abs() % 1000 == 0)).astype("float32")

    # ---- Untitled9-style history features (rebuilt from EARLIER rows only)
    n_g = pd.Series(np.arange(len(df)), dtype="float64")
    s1 = df["amount"].cumsum() - df["amount"]; s2 = (df["amount"] ** 2).cumsum() - df["amount"] ** 2
    m_g = s1 / n_g.replace(0, np.nan); sd_g = np.sqrt((s2 / n_g.replace(0, np.nan) - m_g ** 2).clip(lower=0))
    df["amount_zscore"] = ((df["amount"] - m_g) / sd_g.replace(0, np.nan)).replace([np.inf, -np.inf], np.nan)
    n_v, _, _ = prior_stats(df, ["_v"], "amount"); s_v = df.groupby("_v", sort=False)["amount"].cumsum() - df["amount"]
    df["vendor_transaction_count"] = n_v.astype("float32"); df["vendor_avg_amount"] = s_v / n_v.replace(0, np.nan)
    df["amount_to_vendor_avg_ratio"] = df["amount"] / df["vendor_avg_amount"].replace(0, np.nan)
    n_d, _, _ = prior_stats(df, ["_d"], "amount"); s_d = df.groupby("_d", sort=False)["amount"].cumsum() - df["amount"]
    df["amount_to_department_avg_ratio"] = df["amount"] / (s_d / n_d.replace(0, np.nan)).replace(0, np.nan)
    n_e, _, _ = prior_stats(df, ["_e"], "amount"); s_e = df.groupby("_e", sort=False)["amount"].cumsum() - df["amount"]
    df["employee_transaction_count"] = n_e.astype("float32"); df["employee_avg_amount"] = s_e / n_e.replace(0, np.nan)
    df["_ym"] = ts.dt.to_period("M").astype(str)
    df["vendor_monthly_frequency"] = df.groupby(["_v", "_ym"], sort=False).cumcount().astype("float32")

    # ---- variant C: robust versions (log space, capped counts, explicit "new / missing" flags)
    n_v, m_v, sd_v = prior_stats(df, ["_v"], "_la"); n_e, m_e, sd_e = prior_stats(df, ["_e"], "_la")
    n_d, m_d, sd_d = prior_stats(df, ["_d"], "_la"); n_a, m_a, sd_a = prior_stats(df, ["_all"], "_la")
    n_p, _, _ = prior_stats(df, ["_e", "_v"], "_la"); la = df["_la"]
    df["z_global"] = np.where(n_a >= 50, (la - m_a) / (sd_a + 1e-6), 0.0)
    df["z_vendor"] = np.where(n_v >= 3, (la - m_v) / (sd_v + 0.25), 0.0)
    df["z_employee"] = np.where(n_e >= 3, (la - m_e) / (sd_e + 0.25), 0.0)
    df["z_department"] = np.where(n_d >= 20, (la - m_d) / (sd_d + 0.25), 0.0)
    for c in ["z_global", "z_vendor", "z_employee", "z_department"]: df[c] = df[c].clip(-10, 10)
    df["log_vendor_prior_n"], df["log_employee_prior_n"] = np.log1p(n_v.clip(upper=COUNT_CAP)), np.log1p(n_e.clip(upper=COUNT_CAP))
    df["vendor_is_new"], df["employee_is_new"] = (n_v == 0).astype("float32"), (n_e == 0).astype("float32")
    df["pair_is_new"] = (n_p == 0).astype("float32")
    df["vendor_missing"] = df["vendor_id"].isna().astype("float32"); df["employee_missing"] = df["employee_id"].isna().astype("float32")
    df["log_vendor_freq_30d"] = np.log1p(prior_window_count(df, "_v", 30))
    gap = (ts - df.groupby("_v", sort=False)["_event_time"].shift()).dt.total_seconds() / 86400
    df["log_days_since_vendor"] = np.log1p(gap.fillna(365).clip(lower=0))

    # ---- OCR / receipt features (row-level)
    ocr = pd.to_numeric(df["ocr_extracted_amount"], errors="coerce") if "ocr_extracted_amount" in df else pd.Series(np.nan, index=df.index)
    claim = pd.to_numeric(df["receipt_claimed_amount"], errors="coerce") if "receipt_claimed_amount" in df else pd.Series(np.nan, index=df.index)
    df["ocr_extracted_amount"], df["receipt_claimed_amount"] = ocr, claim
    df["has_receipt"] = (ocr.notna() | claim.notna()).astype("float32")
    df["ocr_mismatch_ratio"] = ((ocr - claim).abs() / claim.replace(0, np.nan)).fillna(0).clip(0, 10)
    df["log_ocr_amount"], df["log_receipt_amount"] = np.log1p(ocr.fillna(0).abs()), np.log1p(claim.fillna(0).abs())
    return df.drop(columns=["_source_order", "_all", "_ym"])

def pit_leakage_test(raw_df, builder=None, cols=None, cutoffs=(0.3, 0.6, 0.85), seed=0):
    """Corrupt every row AFTER a cutoff (amounts x20-100, vendor/employee ids shuffled).
    Point-in-time features of rows BEFORE the cutoff must not change at all."""
    rng = np.random.default_rng(seed)
    ev = pd.to_datetime(raw_df["transaction_date"], errors="coerce")
    tcol = first_existing(raw_df, "transaction_time", "time")
    if tcol: ev = ev + pd.to_timedelta(raw_df[tcol].astype(str), errors="coerce").fillna(pd.Timedelta(0))
    base_raw = raw_df.assign(_t=ev).dropna(subset=["_t"]).sort_values("_t", kind="stable").drop(columns="_t").reset_index(drop=True)
    base_raw = base_raw[pd.to_numeric(base_raw["amount"], errors="coerce").notna()].reset_index(drop=True)   # same rows the builder keeps
    builder, cols = builder or build_features, cols or PIT_COLS
    base = builder(base_raw); worst = 0.0
    for c in cutoffs:
        k = int(len(base) * c); t = base_raw.copy()
        t.loc[k:, "amount"] = pd.to_numeric(t.loc[k:, "amount"], errors="coerce").values * rng.uniform(20, 100, len(t) - k)
        for col in ("vendor_id", "employee_id"):
            t.loc[k:, col] = rng.permutation(t.loc[k:, col].values)
        alt = builder(t)
        a, b = base.loc[:k - 1, cols].astype(float).values, alt.loc[:k - 1, cols].astype(float).values
        worst = max(worst, float(np.abs(np.nan_to_num(a, nan=-1e12) - np.nan_to_num(b, nan=-1e12)).max()))
    return worst < 1e-9, worst

leak_ok, leak_worst = pit_leakage_test(raw)
print(f"Point-in-time features: {'PASS' if leak_ok else 'FAIL'} (max change before cutoff = {leak_worst:.2e})")

def clip_scaled(x): return np.clip(x, -CLIP, CLIP)

def make_preprocessor(num, cat, robust):
    steps = [("impute", SimpleImputer(strategy="median", add_indicator=True)), ("scale", StandardScaler())]
    if robust: steps.append(("clip", FunctionTransformer(clip_scaled, feature_names_out="one-to-one")))
    return ColumnTransformer([("num", Pipeline(steps), num),
                              ("cat", OneHotEncoder(handle_unknown="ignore", min_frequency=5, sparse_output=False), cat)])

FAMILIES = ["AE only", "IF only", "Combined mean", "Combined max", "Legacy 0-1 clipped"]

def robust_z(ref, s):
    """Log-scale score centred/scaled by a clean reference (median/MAD): unclipped, so top scores are not tied."""
    lr, l = np.log(ref + 1e-12), np.log(s + 1e-12)
    med = np.median(lr); mad = np.median(np.abs(lr - med)) * 1.4826 + 1e-9
    return (l - med) / mad

def score_families(ae, iso, x, refs):
    e, f = ae.errors(x), -iso.score_samples(x)
    z_ae, z_if = robust_z(refs["ae"], e), robust_z(refs["if"], f)
    (lo_a, hi_a), (lo_i, hi_i) = refs["ae_q"], refs["if_q"]
    legacy = (0.5 * np.clip((e - lo_a) / max(hi_a - lo_a, 1e-9), 0, 1) + 0.5 * np.clip((f - lo_i) / max(hi_i - lo_i, 1e-9), 0, 1))
    return {"AE only": z_ae, "IF only": z_if, "Combined mean": (z_ae + z_if) / 2,
            "Combined max": np.maximum(z_ae / refs["t_ae"], z_if / refs["t_if"]), "Legacy 0-1 clipped": legacy}, e, f

def make_refs(ae, iso, x_train, x_val_normal):
    r_ae, r_if = ae.errors(x_val_normal), -iso.score_samples(x_val_normal)
    q99 = lambda z: max(float(np.quantile(z, 0.99)), 1e-3)
    return {"ae": r_ae, "if": r_if, "t_ae": q99(robust_z(r_ae, r_ae)), "t_if": q99(robust_z(r_if, r_if)),
            "ae_q": tuple(np.quantile(ae.errors(x_train), [0.01, 0.995])),
            "if_q": tuple(np.quantile(-iso.score_samples(x_train), [0.01, 0.995]))}

def best_f1_threshold(y, s):
    p, r, t = precision_recall_curve(y, s)
    f1 = 2 * p[:-1] * r[:-1] / np.clip(p[:-1] + r[:-1], 1e-12, None)
    return float(t[int(np.argmax(f1))])

def threshold_metrics(y, s, thr, prefix):
    pred = (s >= thr).astype(int)
    return {f"{prefix}_precision": precision_score(y, pred, zero_division=0), f"{prefix}_recall": recall_score(y, pred, zero_division=0),
            f"{prefix}_f1": f1_score(y, pred, zero_division=0), f"{prefix}_alert_rate": pred.mean(),
            f"{prefix}_normal_fpr": float((s[y == 0] >= thr).mean())}

            "thr_f1": thr_f1, "thr_q": thr_q, **threshold_metrics(y_test, s_test, thr_f1, "f1thr"),
            **threshold_metrics(y_test, s_test, thr_q, "qthr")}

def psi(ref, cur, bins=10, eps=1e-4):
    ref, cur = np.asarray(ref, float), np.asarray(cur, float); ref, cur = ref[~np.isnan(ref)], cur[~np.isnan(cur)]
    if len(np.unique(ref)) <= 5:
        cats = np.union1d(np.unique(ref), np.unique(cur))
        p, q = np.array([(ref == c).mean() for c in cats]), np.array([(cur == c).mean() for c in cats])
    else:
        edges = np.unique(np.quantile(ref, np.linspace(0, 1, bins + 1))); edges[0], edges[-1] = -np.inf, np.inf
        p, q = np.histogram(ref, edges)[0] / len(ref), np.histogram(cur, edges)[0] / len(cur)
    p, q = np.clip(p, eps, None), np.clip(q, eps, None); return float(np.sum((q - p) * np.log(q / p)))
